CSS tokenek — másolható kód

Minden HTML fájl inline <style> blokkjának elején szerepel ez a :root definíció.

/* Fonts */ @font-face{font-family:'Planer';src:url('fonts/Planer-Regular.woff2') format('woff2');font-weight:400;font-display:swap;} @font-face{font-family:'Planer';src:url('fonts/Planer-DemiBold.woff2') format('woff2');font-weight:600;font-display:swap;} /* Design tokens */ :root{ --blue:#1e3184; --blue-dark:#162560; --blue-light:#e8ecf7; --gray-50:#f9fafb; --gray-100:#f3f4f6; --gray-200:#e5e7eb; --gray-500:#6b7280; --gray-600:#6e7681; --gray-700:#374151; --gray-900:#111827; --dark:#0f172a; --radius:12px; --shadow-sm:0 1px 3px rgba(0,0,0,.08); --shadow-md:0 4px 16px rgba(0,0,0,.10); --shadow-lg:0 10px 40px rgba(0,0,0,.14); --transition:0.25s ease; }
